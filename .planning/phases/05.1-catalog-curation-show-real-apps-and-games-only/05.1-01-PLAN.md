    ```cpp
        QStringLiteral("control panel"), QStringLiteral("command prompt"),
        QStringLiteral("windows powershell"), QStringLiteral("registry editor"),
        QStringLiteral("event viewer"), QStringLiteral("task scheduler"),
        QStringLiteral("disk cleanup"), QStringLiteral("services"),

    };
    const QStringList kSystemUtilExes = {
        QStringLiteral("cmd.exe"), QStringLiteral("powershell.exe"),
        QStringLiteral("regedit.exe"), QStringLiteral("control.exe"),
        QStringLiteral("msconfig.exe"), QStringLiteral("services.msc"),
        QStringLiteral("eventvwr.msc"), QStringLiteral("taskmgr.exe"),
        QStringLiteral("cleanmgr.exe"), QStringLiteral("wsl.exe"),
    };
    // Bucket 2: dev toolchains (~50). Native separators (parseLnk contract).
    const QStringList kPathFragments = {
        QStringLiteral("\\msys"), QStringLiteral("\\qt\\"), QStringLiteral("\\python"),
        QStringLiteral("\\nodejs"), QStringLiteral("\\git\\"),
        QStringLiteral("\\visual studio"), QStringLiteral("\\windows kits"),
        QStringLiteral("\\cmake"), QStringLiteral("\\go"), QStringLiteral("\\jdk"),
        QStringLiteral("\\jre"), QStringLiteral("\\openjdk"), QStringLiteral("\\openssl"),
        QStringLiteral("\\inno setup"), QStringLiteral("\\nsis"),
    };
    // Deliberate non-matches (Pitfall 4): "Windows Terminal", "Microsoft
    // Store", "Visual Studio Code" — no token/path rule may touch them
    // (VS Code's folder is "Microsoft VS Code", not "Visual Studio").

    // Word-boundary normalization: fold case, collapse every non-alphanumeric
    // run to a single space. Multi-word tokens match as whole sequences.
    QString normalized(const QString &s)
    {
        QString out;
        out.reserve(s.size());
        bool lastWasSpace = true;
        for (const QChar &c : s) {
            if (c.isLetterOrNumber()) { out.append(c.toCaseFolded()); lastWasSpace = false; }
            else if (!lastWasSpace) { out.append(QLatin1Char(' ')); lastWasSpace = true; }
        }
        return out;
    }
    bool containsWord(const QString &normalized, const QString &token)
    {
        return normalized == token
            || normalized.startsWith(token + QLatin1Char(' '))
            || normalized.endsWith(QLatin1Char(' ') + token)
            || normalized.contains(QLatin1Char(' ') + token + QLatin1Char(' '));
    }
    } // namespace

    bool CurationRules::matches(const AppEntry &e)
    {
        // CUR-04 belt-and-braces: added executables (Source::File) are never
        // curated — the escape hatch holds even if a caller forgets the guard.
        if (e.source == AppEntry::Source::File)
            return false;
        const QString name = normalized(e.displayName);
        for (const QString &p : kNamePrefixes)
            if (name.startsWith(p))
                return true;
        for (const QString &t : kNameTokens)
            if (containsWord(name, t))
                return true;
        for (const QString &u : kSystemUtilNames)
            if (containsWord(name, u))
                return true;
        // Path-based rules apply to Lnk rows only (UWP rows have no path).
        if (!e.targetPath.isEmpty()) {
            const QString fileName = QFileInfo(e.targetPath).fileName().toCaseFolded();
            for (const QString &x : kSystemUtilExes)
                if (fileName == x)
                    return true;
            const QString foldedPath = e.targetPath.toCaseFolded();
            for (const QString &f : kPathFragments)
                if (foldedPath.contains(f))
                    return true;
        }
        return false;
    }
    ```

    ```cpp
    #include "core/CurationStore.h"

    namespace {

    const QString kHiddenGroup = QStringLiteral("curationHidden");
    const QString kShownGroup  = QStringLiteral("curationShown");

    // PATTERNS P1 — SettingsStore.cpp:11-17 VERBATIM.
    QSettings makeSettings(const QString &settingsPath)
    {
        if (settingsPath.isEmpty())
            return QSettings(QSettings::IniFormat, QSettings::UserScope,
                             QStringLiteral("TID"), QStringLiteral("wisp"));
        return QSettings(settingsPath, QSettings::IniFormat);
    }

    } // namespace

    CurationStore::CurationStore(const QString &settingsPath)
        : m_settings(makeSettings(settingsPath))
    {
    }

    void CurationStore::hide(const QString &id)
    {
        if (id.isEmpty())
            return; // identity contract — never write a key without an id
        m_settings.setValue(kHiddenGroup + QLatin1Char('/') + id, 1);
        // Last action wins: a fresh hide cancels a previous show.
        m_settings.remove(kShownGroup + QLatin1Char('/') + id);
        m_settings.sync(); // PATTERNS P2 — sync after EVERY write
    }

    void CurationStore::show(const QString &id)
    {
        if (id.isEmpty())
            return;
        m_settings.setValue(kShownGroup + QLatin1Char('/') + id, 1);
        m_settings.remove(kHiddenGroup + QLatin1Char('/') + id);
        m_settings.sync();
    }

    QSet<QString> CurationStore::hiddenIds() const
    {
        const QString prefix = kHiddenGroup + QLatin1Char('/');
        const QStringList keys = m_settings.allKeys();
        QSet<QString> ids;
        for (const QString &key : keys)
            if (key.startsWith(prefix))
                ids.insert(key.mid(prefix.size()));
        return ids;
    }

    QSet<QString> CurationStore::shownIds() const
    {
        const QString prefix = kShownGroup + QLatin1Char('/');
        const QStringList keys = m_settings.allKeys();
        QSet<QString> ids;
        for (const QString &key : keys)
            if (key.startsWith(prefix))
                ids.insert(key.mid(prefix.size()));
        return ids;
    }
    ```
