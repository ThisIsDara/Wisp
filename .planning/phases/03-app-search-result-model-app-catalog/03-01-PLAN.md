```cpp
namespace FuzzyMatcher {
struct MatchRange { int start; int length; };   // contiguous run of matched chars
struct Result { int score; QVector<MatchRange> ranges; };
// Empty query → { score: 0, ranges: {} }. Case-insensitive. Ladder:
// exact > name prefix > word-boundary start > any subsequence; camelCase
// bonus (uppercase following lowercase = boundary). No cutoff (D-06):
// any subsequence match scores > 0.
Result score(const QString &query, const QString &displayName);
}
```

```cpp
class ResultsModel : public QAbstractListModel {
    Q_OBJECT
    Q_PROPERTY(QString query READ query NOTIFY queryChanged)  // consumed by 03-05 empty-state copy
public:
    enum Roles { DisplayNameRole = Qt::UserRole + 1, SubtitleRole, MatchRangesRole, AumidRole };
    explicit ResultsModel(QObject *parent = nullptr);
    // Full catalog snapshot (03-03 signals → this). Resets query to "" and selection to 0.
    void setEntries(QVector<AppEntry> entries);
    // Query → filter+rank via FuzzyMatcher::score, sort score desc then displayName asc (D-05).
    // Empty query → all entries, alphabetical case-insensitive (D-01), selection index 0 (D-02).
    Q_INVOKABLE void setQuery(const QString &query);
    QString query() const;   // for the 03-05 "No results for \"{query}\"" interpolation
    int rowCount(const QModelIndex &parent = {}) const override;
    QVariant data(const QModelIndex &idx, int role) const override;
    // Selection (LAUN-05): clamped to [0, rowCount-1]; delta ±1 for ↑/↓,
    // ±kVisibleRows (7) for PageUp/PageDown; Home=0 / End=count-1.
    Q_INVOKABLE int selectedIndex() const;
    Q_INVOKABLE void moveSelection(int delta);
    Q_INVOKABLE void selectIndex(int index);
    // D-12 freeze seed: value copy of the entry at the CURRENT selection
    // (03-04's launchSelected() calls this at keypress).
    AppEntry snapshotSelected() const;
signals:
    void queryChanged(const QString &query);
private:
    QVector<AppEntry> m_entries;  // always sorted for display
    QString m_query;
    QVector<int> m_order;         // permutation of m_entries for current query
    int m_selected = 0;
    static constexpr int kVisibleRows = 7;   // 640×400 shell ≈ 7 rows of 44px (UI-SPEC geometry)
};
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| query string → matcher | User-typed text crosses into scoring logic every keystroke; can be arbitrarily long |
| matcher → QML delegate | Match ranges derived from strings are passed to QML rendering (Phase 5 visuals) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01-01 | D | FuzzyMatcher::score | mitigate | Linear O(query×name) scan only — no regex, no recursion, no allocation per char; query longer than name short-circuits to no-match; perf smoke test asserts &lt;5ms for 500 names (D-06 budget) |
| T-03-01-02 | S | QString length abuse | accept | Query length bounded by QML TextField; matcher degrades gracefully (linear) — same-user local input, no privilege boundary |
| T-03-01-03 | I | match ranges indexing | mitigate | Ranges computed in one place at query time, always clamped to [0, name.length()] (Test 5 asserts in-bounds); QML Phase 5 renders them as text formatting only, never as indexes into other buffers |
| T-03-01-04 | E | AppEntry display data | accept | Metadata only (names/paths — same-user data); rendering is Qt Quick text, no HTML/script interpretation |
</threat_model>
